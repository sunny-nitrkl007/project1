**********************************************************************************
Attributes : when DSI is avail - add ( dsi / dsi used / DSI ) in attributes
             when DSI is unavail - not to add (non dsi/dsi unvail) in attributes 
	     when data is LSB - add swap bytes in attributes
             when data is MSB -  not to add (unswap/swap) in attributes
	     when data is signed - add signed in attributes
             when data is unsigned - not to add (signed/unsigned) in attributes

ex  Attributes  : "" // It means dsi unavail, MSB, unsigned

RequestType  :  typedef enum
		{
		   J1939_SPB_REQUEST = 0,
		   J1939_PGB_REQUEST,
		   J1939_ADT_REQUEST,
		   J1939_READ_POLL,
		   J1939_MAP_REQUEST,
		   J1939_INVALID_REQ_TYPE
		} J1939ParamReqType_t;

VarLengthDataType :

 //VarLengthParamType values are used to reference entries in the m_VarLengthPool.  Always add new
      //enum values to the end of the list because the integer values of the enum appear in a json
      //data initialization file for BmiCdlAppCfgTbl.
      enum VarLengthParamType
      {
         NOT_APPLICABLE,
         GPS,
         SPL_PID_REV_DATA_BYTE_ORDER, // use this for speacial pid's to reverse data (ex pid:F5D9)
         TIME_ZONE,
         PRODUCT_ID,
         PGN, // Use for Public PGN Data
         UNSUPPORTED
      };

ParameterType  : 
// From file  cdl2_struct.h
typedef enum
{
  CDL2_U8_DSI = 0,         /* Unsigned Byte + Data Status ID */
  CDL2_I8_DSI,             /* Signed Byte + Data Status ID */
  CDL2_U16_DSI,            /* Unsigned Word + Data Status ID */
  CDL2_I16_DSI,            /* Signed Word + Data Status ID */
  CDL2_U32_DSI,            /* Unsigned Long + Data Status ID */
  CDL2_I32_DSI,            /* Signed Long + Data Status ID */
  CDL2_U8_NO_DSI,          /* Unsigned Byte + ** NO ** Data Status ID */
  CDL2_I8_NO_DSI,          /* Signed Byte + ** NO ** Data Status ID */
  CDL2_U16_NO_DSI,         /* Unsigned Word + ** NO ** Data Status ID */
  CDL2_I16_NO_DSI,         /* Signed Word + ** NO ** Data Status ID */
  CDL2_U32_NO_DSI,         /* Unsigned Long + ** NO ** Data Status ID */
  CDL2_I32_NO_DSI,         /* Signed Long + ** NO ** Data Status ID */
  CDL2_VAR_NO_DSI          /* Variable length, NO Data Status Checking Defined */
} cdl2_parameter_type_e;


Cid : Component ID  for ECM


*******************************************************************************

** PGN       J1939 PGN identifying this message.
**
** Handler Index 	The index of each item in the following array must match "Handler Index" in the JSON file
**			const scl_j1939_rx_mpfn_parser_t* BmiJ1939DataHandlers::m_pgnHandlers[] = {
**     			BmiJ1939PGNDefault,                 // Index 0 - User defined handler 
**      		scl_j1939_cat_ppgn_hdlr             // Index 1 - CSNS Handler
**			};
**
** Priority      The priority of the mesage when requesting this PGN.
**
**
** Source Address The source address (the network address of the ECM
**                from where the PGN came) of this PGN.
**
** Source Name       The J1939 SHORTHAND NAME of the device that originated this PGN.
**                   The name field consists of Arbitrary Address Capable,Industry Group,
**                   Vehicle System Instance,Vehicle System,Reserved,Function,
**                   Function Instance,ECU Instance.                      
**
** Control Bitfield    Initial value that indicates how the message is received,
**                	e.g., if the lost timer is applicable.
**    Allowed values:
**    00 - Inactive (SCL_J1939_RX_MSG_CBF)
**    01 - Enabled (SCL_J1939_RX_MSG_CBF_ENABLED)
**    02 - Use time out value (SCL_J1939_RX_MSG_CBF_LOST_TIMER_APPLICABLE)
**
** Timeout(ms)        The amount of time before a message is assumed lost.
**
** Cid           Component ID for ECM, needed for health monitoring
**
** SPN List      Comma separated string of SPNs, used for health monitoring
